-r requirements.txt

# WSGI-сервер. Django development server не рассчитан на нагрузку и сам
# об этом предупреждает: он однопоточный по замыслу и не переживает
# ни медленного клиента, ни отвалившегося соединения.
gunicorn~=23.0

# ASGI-сервер для одного отдельного процесса — потока событий CRM (SSE) и
# ожидания очереди ботом. Долгое соединение в gunicorn держало бы
# синхронный воркер целиком; здесь оно стоит корутину. Обычный API
# остаётся на gunicorn. Без [standard]: uvloop и httptools не нужны
# процессу с десятком долгих соединений.
uvicorn~=0.54.0
